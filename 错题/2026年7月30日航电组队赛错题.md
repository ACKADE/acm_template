# TREE

## 题目描述

给定一个长度为 $n$ 的排列 $a_1, a_2, \dots, a_n$。

对于一个非空序列 $b_1, b_2, \dots, b_k$，它的小根笛卡尔树是一棵满足以下条件的二叉树：

- 结点为序列中的 $k$ 个位置；
- 中序遍历依次得到位置 $1, 2, \dots, k$；
- 每个结点对应的值都小于其儿子对应的值。

因为序列中的数互不相同，所以它的小根笛卡尔树唯一。

树根的深度定义为 $1$，其余结点的深度等于父亲深度加 $1$。树的高度是所有结点深度的最大值。

有 $q$ 次询问。每次给定一个区间 $[l, r]$，独立地取出序列 $a_l, a_{l+1}, \dots, a_r$，建立它的小根笛卡尔树，并求出这棵树的高度。

- $1 \le T \le 10$；
- $1 \le n, q \le 2 \times 10^5$；
- $a_1, a_2, \dots, a_n$ 是 $1, 2, \dots, n$ 的一个排列；
- $1 \le l \le r \le n$；
- 所有测试数据的 $n$ 之和不超过 $4 \times 10^5$；
- 所有测试数据的 $q$ 之和不超过 $4 \times 10^5$。

### 输入格式

输入包含多组测试数据。第一行包含一个整数 $T$，表示测试数据组数。

对于每组测试数据：

- 第一行包含两个整数 $n, q$；
- 第二行包含 $n$ 个整数 $a_1, a_2, \dots, a_n$；
- 接下来 $q$ 行，每行包含两个整数 $l, r$，表示一次询问。

### 输出格式

对于每次询问，输出一行一个整数，表示对应区间的小根笛卡尔树高度。

## 题解

记整段排列的小根笛卡尔树为 $$T$$ ，结点编号仍为原下标， $$h_{u}$$ 表示 $$u$$ 的子树高度，约定 $$h_{0} = 0$$ 。笛卡尔树的中序遍历就是原序列，因此

$$m = \mathrm{LCA}(l,r)$$

是区间 $$[l,r]$$ 的根。左半部分是 $$l c_{m}$$ 子树的一段后缀，右半部分是 $$r c_{m}$$ 子树的一段前缀。

设 $$\mathrm{Pref}(u,x)$$ 表示 $$u$$ 子树中从最左端到 $$x$$ 这一段的高度，初值为

$$t = 1 + h_{l c_{x}}.$$

从 $$x$$ 向 $$u$$ 上跳。若当前点是父亲 $$p$$ 的右儿子，还要加入 $$p$$ 及其左子树，此时

$$t\leftarrow \max (1 + h_{l c_{p}},t + 1).$$

否则无需修改。后缀完全对称，初值为 $$1 + h_{r c_{x}}$$ ；当前点是父亲 $$p$$ 的左儿子时，执行

$$t\leftarrow \max (1 + h_{r c_{p}},t + 1).$$

所有转移都形如

$$f(t) = \max (A,t + B).$$

用二元组 $$(A,B)$$ 表示。若merge(f,g)表示 $$f\circ g$$ ，则

$$\mathrm{merge}\big((A,B),(C,D)\big) = \big(\max (A,C + B),B + D\big).$$

倍增维护每段祖先路径上的函数复合。从后代向祖先查询时，新取出的上方路径放在已有复合函数的外层。

最后令

$$L = 
\begin{cases} 
\text{Stuff}(lc_m, l), & l < m, \\ 
0, & l = m, 
\end{cases} 
\quad R = 
\begin{cases} 
\text{Pref}(rc_m, r), & m < r, \\ 
0, & m = r, 
\end{cases}$$

答案为

$$1 + \max (L,R).$$

时间复杂度 $$O(n\log n)$$ 。

```c++
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const int N = 4e5 + 5;
const int LOGN = 20;

int n, q;
int a[N];
int L[N], R[N];
int lc[N], rc[N];
int depth_L[N], depth_R[N];
int h_left[N], h_right[N], h_total[N];
int val_L[N], val_R[N];

// ST 表 RMQ 查询区间最小值下标
int st[LOGN][N];
int lg[N];

inline int get_min_pos(int l, int r) {
    int k = lg[r - l + 1];
    int p1 = st[k][l];
    int p2 = st[k][r - (1 << k) + 1];
    return a[p1] < a[p2] ? p1 : p2;
}

// 树上倍增表 (TR 与 TL 树)
int up_R[LOGN][N], max_R[LOGN][N], dep_TR[N];
int up_L[LOGN][N], max_L[LOGN][N], dep_TL[N];

void dfs_ct(int u) {
    if (!u) return;

    if (lc[u]) {
        depth_L[lc[u]] = depth_L[u];
        depth_R[lc[u]] = depth_R[u] + 1;
        dfs_ct(lc[u]);
        h_left[u] = h_total[lc[u]];
    } else {
        h_left[u] = 0;
    }

    if (rc[u]) {
        depth_L[rc[u]] = depth_L[u] + 1;
        depth_R[rc[u]] = depth_R[u];
        dfs_ct(rc[u]);
        h_right[u] = h_total[rc[u]];
    } else {
        h_right[u] = 0;
    }

    h_total[u] = 1 + max(h_left[u], h_right[u]);
}

inline int query_R(int u, int p) {
    int dist = dep_TR[u] - dep_TR[p];
    int res = 0;
    for (int i = LOGN - 1; i >= 0; i--) {
        if ((dist >> i) & 1) {
            res = max(res, max_R[i][u]);
            u = up_R[i][u];
        }
    }
    return res;
}

inline int query_L(int u, int p) {
    int dist = dep_TL[u] - dep_TL[p];
    int res = 0;
    for (int i = LOGN - 1; i >= 0; i--) {
        if ((dist >> i) & 1) {
            res = max(res, max_L[i][u]);
            u = up_L[i][u];
        }
    }
    return res;
}

void solve() {
    cin >> n >> q;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        lc[i] = rc[i] = 0;
    }

    // 1. 单调栈求 L_i 和 R_i
    vector<int> stk;
    for (int i = 1; i <= n; i++) {
        while (!stk.empty() && a[stk.back()] > a[i]) stk.pop_back();
        L[i] = stk.empty() ? 0 : stk.back();
        stk.push_back(i);
    }
    stk.clear();
    for (int i = n; i >= 1; i--) {
        while (!stk.empty() && a[stk.back()] > a[i]) stk.pop_back();
        R[i] = stk.empty() ? 0 : stk.back();
        stk.push_back(i);
    }

    // 2. 构建全局笛卡尔树
    int root = 0;
    for (int i = 1; i <= n; i++) {
        if (L[i] == 0 && R[i] == 0) root = i;
        else if (L[i] == 0) lc[R[i]] = i;
        else if (R[i] == 0) rc[L[i]] = i;
        else {
            if (a[L[i]] > a[R[i]]) rc[L[i]] = i;
            else lc[R[i]] = i;
        }
    }

    // 3. DFS 计算全局深度与左右子树高度
    depth_L[root] = depth_R[root] = 0;
    dfs_ct(root);

    // 【修正位置】：val_L 搭配 h_left，val_R 搭配 h_right
    for (int i = 1; i <= n; i++) {
        val_L[i] = h_left[i] + depth_L[i] + 1;
        val_R[i] = h_right[i] + depth_R[i] + 1;
    }

    // 4. 构建 ST 表
    for (int i = 1; i <= n; i++) st[0][i] = i;
    for (int j = 1; j < LOGN; j++) {
        for (int i = 1; i + (1 << j) - 1 <= n; i++) {
            int p1 = st[j - 1][i];
            int p2 = st[j - 1][i + (1 << (j - 1))];
            st[j][i] = (a[p1] < a[p2]) ? p1 : p2;
        }
    }

    // 5. TR 树倍增初始化
    dep_TR[0] = 0;
    for (int i = n; i >= 1; i--) {
        up_R[0][i] = R[i];
        max_R[0][i] = val_R[i];
        dep_TR[i] = dep_TR[R[i]] + 1;
    }
    for (int j = 1; j < LOGN; j++) {
        for (int i = 1; i <= n; i++) {
            up_R[j][i] = up_R[j - 1][up_R[j - 1][i]];
            max_R[j][i] = max(max_R[j - 1][i], max_R[j - 1][up_R[j - 1][i]]);
        }
    }

    // 6. TL 树倍增初始化
    dep_TL[0] = 0;
    for (int i = 1; i <= n; i++) {
        up_L[0][i] = L[i];
        max_L[0][i] = val_L[i];
        dep_TL[i] = dep_TL[L[i]] + 1;
    }
    for (int j = 1; j < LOGN; j++) {
        for (int i = 1; i <= n; i++) {
            up_L[j][i] = up_L[j - 1][up_L[j - 1][i]];
            max_L[j][i] = max(max_L[j - 1][i], max_L[j - 1][up_L[j - 1][i]]);
        }
    }

    // 7. 回答查询
    while (q--) {
        int l, r;
        cin >> l >> r;
        int p = get_min_pos(l, r);
        int ans_L = 0, ans_R = 0;

        if (l < p) ans_L = query_R(l, p) - depth_R[p];
        if (r > p) ans_R = query_L(r, p) - depth_L[p];

        cout << max({1, ans_L, ans_R}) << "\n";
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    lg[1] = 0;
    for (int i = 2; i < N; i++) lg[i] = lg[i / 2] + 1;

    int T = 1;
    if (cin >> T) {
        while (T--) solve();
    }
    return 0;
}
```

# Median Shuffle

## 题目描述

本题开启 Special Judge。

给定一个奇数 $n$。存在一个长度为 $n$ 的排列 $a$。

令 $m = \frac{n+1}{2}$。

构造一个长度为 $m$ 的序列 $b$。对于每个 $1 \le i \le m$，定义：
$$b_i = \operatorname{median}(a_1, a_2, \dots, a_{2i-1}).$$
其中 $\operatorname{median}$ 为取中位数操作，长度为 $2i-1$ 的序列的中位数是将其从小到大排序后的第 $i$ 个元素。

将序列 $a$ 和序列 $b$ 拼接，再将得到的序列任意打乱，得到一个长度为 $n+m = \frac{3n+1}{2}$ 的序列 $c$。

现在给定 $n$ 和打乱后的序列 $c$，请构造任意一个可能的排列 $a$。如果不存在满足条件的排列，输出 `-1`。

序列 $c$ 中元素的顺序不包含任何额外信息。

样例中，排列可以取为 $[2,1,3]$。长度为 $1$ 和 $3$ 的奇数前缀中位数均为 $2$，因此 $a$ 与中位数序列拼接后的多重集合恰好等于输入的多重集合。

样例输出不是唯一的。

- $1 \le T \le 10$；
- $1 \le n \le 3 \times 10^5$，且 $n$ 为奇数；
- $1 \le c_i \le n$；
- 所有测试数据的 $n$ 之和不超过 $6 \times 10^5$。

### 输入格式

输入包含多组测试数据。第一行包含一个整数 $T$ ($1 \le T \le 10$)，表示测试数据组数。

对于每组测试数据：
- 第一行包含一个奇数 $n$ ($1 \le n \le 3 \times 10^5$)；
- 第二行包含 $\frac{3n+1}{2}$ 个整数 $c_1, c_2, \dots, c_{(3n+1)/2}$ ($1 \le c_i \le n$)。

### 输出格式

对于每组测试数据：
- 如果不存在满足条件的排列，输出一行 `-1`；
- 否则，输出一行 $n$ 个整数 $a_1, a_2, \dots, a_n$，表示任意一个满足条件的排列。如果存在多组合法答案，输出任意一组即可。

#### 样例输入

```
1
3
2 1 2 3 2
```

#### 样例输出

```c++
2 1 3
```

## 题解

记值 $$v$$ 在 $$c$$ 中出现了 $$C_{v}$$ 次。

因为 $$a$$ 是一个排列,所以每个值在 $$a$$ 中恰好出现一次。因此,值 $$v$$ 需要在 $$b$$ 中出现

$$w_{v} = C_{v} - 1$$

次。如果某个 $$C_{v} = 0$$ ,则无解。

在所有 $$C_{v}\geq 1$$ 时,由输入长度可知

$$\sum_{v = 1}^{n}w_{v} = m.$$

问题转化为:能否构造一个排列,使值 $$v$$ 恰好作为奇数前缘中位数出现 $$w_{v}$$ 次。

充要条件为:对于每个 $$0\leq d< m$$

$$\sum_{v = m - d}^{m + d}w_{v}\geq d + 1.$$

**必要性**

从完整集合开始,每次逆向删除排列末尾的两个元素。删除 $$2t$$ 个元素后,剩下的集合就是长度为 $$n - 2t$$ 的奇数前级所包含的元素,共有

$$2(m - t) - 1$$

个。设这个集合的中位数为 $$p_{t}$$ 。中位数是第 $$m - t$$ 小的元素,所以集合中至少有 $$m - t$$ 个值不大于 $$p_{t}$$ ,也至少有 $$m - t$$ 个值不小于 $$p_{t}$$ 。

$$m - t\leq p_{t}\leq m + t.$$

对于 $$t = 0,1,\ldots ,d$$ ,这 $$d + 1$$ 个时刻的中位数全部属于

$$[m - d,m + d].$$

**充分性 (构造)**

令 $$r_{v}$$ 表示值 $$v$$ 还需要作为中位数出现多少次,初始时 $$r_{v} = w_{v}$$ 。

设当前集合中还有 $$2k - 1$$ 个值，按大小写成

$$x_{- (k - 1)}< \dots < x_{- 1}< x_{0}< x_{1}< \dots < x_{k - 1},$$

其中 $$p = x_{0}$$ 是当前中位数。维护以下不变量：

$$\sum_{i = -d}^{d}r_{x_{i}}\geq d + 1\qquad (0\leq d< k),$$

并且剩余需求总和恰好为 $$k$$ 。

取 $$d = 0$$ 可知 $$r_{p} > 0$$ 。每轮先让当前中位数 $$p$$ 出现一次，即执行

$$r_{p}\leftarrow r_{p} - 1.$$

若还需继续删除，则只删除需求已经变为0的值。根据 $$r_{p}$$ 是否仍为正，执行以下操作：


<table>使用p后的情况删除的两个值新中位数r_p&amp;gt;0p左右两侧最近的零需求值仍为pr_p=0,左邻需求为正p和右侧最近的零需求值变为左邻r_p=0,否则左侧最近的零需求值和p变为右邻</table>

根据

$$\sum_{i = -d}^{d}r_{x_{i}}\geq d + 1\qquad (0\leq d< k),$$

得知若 $$r_{p} = 0$$ 左右邻一定有一个非零，且按照上述操作后，下一轮上述等式依旧成立。归纳即可。实现是平凡的。

```c++
#include <bits/stdc++.h>
using namespace std;

class BIT {
    int n;
    vector<int> tree;
public:
    BIT(int sz) : n(sz), tree(sz + 1, 0) {}
    void add(int idx, int delta) {
        for (; idx <= n; idx += idx & -idx)
            tree[idx] += delta;
    }
    int sum(int idx) {
        int s = 0;
        for (; idx > 0; idx -= idx & -idx)
            s += tree[idx];
        return s;
    }
    // 找到第 k 个存在的元素 (k >= 1)
    int find_kth(int k) {
        int idx = 0;
        for (int bit = 1 << (20); bit; bit >>= 1) {
            int nxt = idx + bit;
            if (nxt <= n && tree[nxt] < k) {
                k -= tree[nxt];
                idx = nxt;
            }
        }
        return idx + 1;
    }
};

void solve() {
    int n;
    cin >> n;
    int total = (3 * n + 1) / 2;
    vector<int> c(total);
    vector<int> C(n + 1, 0);
    for (int i = 0; i < total; ++i) {
        cin >> c[i];
        C[c[i]]++;
    }

    // 检查每个值至少出现一次
    for (int v = 1; v <= n; ++v) {
        if (C[v] == 0) {
            cout << "-1\n";
            return;
        }
    }

    int m = (n + 1) / 2;
    vector<int> w(n + 1);
    for (int v = 1; v <= n; ++v) {
        w[v] = C[v] - 1;
    }

    // 验证充要条件
    vector<int> pref(n + 1, 0);
    for (int v = 1; v <= n; ++v)
        pref[v] = pref[v - 1] + w[v];
    for (int d = 0; d < m; ++d) {
        int L = m - d, R = m + d;
        int sum_w = pref[R] - pref[L - 1];
        if (sum_w < d + 1) {
            cout << "-1\n";
            return;
        }
    }

    // 初始化 BIT 和需求数组
    BIT bit(n);
    for (int v = 1; v <= n; ++v)
        bit.add(v, 1);
    vector<int> r = w;
    set<int> zero;
    for (int v = 1; v <= n; ++v)
        if (r[v] == 0) zero.insert(v);

    vector<pair<int, int>> deletions;
    int k = m;  // 当前剩余 2k-1 个元素

    while (k > 1) {
        int p = bit.find_kth(k); // 当前中位数
        r[p]--;
        if (r[p] == 0) zero.insert(p);

        if (r[p] > 0) {
            // 删除左右最近的零需求元素
            auto it = zero.lower_bound(p);
            // it 指向 >= p 的第一个零需求
            int u = *prev(it);   // 左侧最近零需求 (必存在)
            int v;
            if (it != zero.end() && *it == p)
                v = *next(it);   // p 自身是零需求, 右侧取下一个
            else
                v = *it;         // p 不是零需求, 右侧就是这个
            deletions.emplace_back(u, v);
            bit.add(u, -1); bit.add(v, -1);
            zero.erase(u); zero.erase(v);
        } else { // r[p] == 0
            int rank_p = bit.sum(p);
            int x_left = 0, x_right = 0;
            if (rank_p > 1) x_left = bit.find_kth(rank_p - 1);
            if (rank_p < 2 * k - 1) x_right = bit.find_kth(rank_p + 1);
            if (x_left != 0 && r[x_left] > 0) {
                // 删除 p 和右侧最近的零需求
                auto it = zero.lower_bound(p);
                int v = *next(it);  // p 在 zero 中, 下一个一定存在
                deletions.emplace_back(p, v);
                bit.add(p, -1); bit.add(v, -1);
                zero.erase(p); zero.erase(v);
            } else {
                // 删除左侧最近的零需求和 p
                auto it = zero.lower_bound(p);
                int u = *prev(it);
                deletions.emplace_back(u, p);
                bit.add(u, -1); bit.add(p, -1);
                zero.erase(u); zero.erase(p);
            }
        }
        --k;
    }

    int center = bit.find_kth(1);
    vector<int> a;
    a.push_back(center);
    // 逆序取出删除对，依次追加到序列末尾
    for (int i = (int)deletions.size() - 1; i >= 0; --i) {
        a.push_back(deletions[i].first);
        a.push_back(deletions[i].second);
    }

    for (int i = 0; i < n; ++i) {
        cout << a[i] << (i + 1 == n ? '\n' : ' ');
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T;
    cin >> T;
    while (T--) {
        solve();
    }
    return 0;
}
```
